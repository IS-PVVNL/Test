// --- RELIABLE MOBILE CAMERA ENGINE --- document.getElementById('scanBtn').addEventListener('click', async () => { const readerDiv = document.getElementById('reader'); if (!isScanning) { // 1. Pehle DOM me display show karein taaki container ko width/height mile readerDiv.classList.remove('hidden'); readerDiv.style.display = 'block'; if (!html5QrCode) { html5QrCode = new Html5Qrcode("reader"); } const scanConfig = { fps: 15, qrbox: (viewWidth, viewHeight) => { const size = Math.floor(Math.min(viewWidth, viewHeight) * 0.75); return { width: size, height: size }; }, experimentalFeatures: { useBarCodeDetectorIfSupported: true } }; try { // Sabse pehle device ke cameras list karein const devices = await Html5Qrcode.getCameras(); if (devices && devices.length > 0) { // Back/Rear camera dhoondein let backCam = devices.find(c => c.label.toLowerCase().includes('back') || c.label.toLowerCase().includes('rear') || c.label.toLowerCase().includes('environment') ); // Agar specific label na mile toh aakhri camera select karein (phones me hamesha back camera last index hota hai) const cameraId = backCam ? backCam.id : devices[devices.length - 1].id; await html5QrCode.start( cameraId, scanConfig, (decodedText) => { const cleanVal = cleanScanText(decodedText); document.getElementById('manualInput').value = cleanVal; searchRecord(cleanVal); stopScanner(); }, (err) => {} ); isScanning = true; document.getElementById('scanBtn').innerText = "Stop Scanner"; } else { // Fallback agar getCameras device list na de await html5QrCode.start( { facingMode: "environment" }, scanConfig, (decodedText) => { const cleanVal = cleanScanText(decodedText); document.getElementById('manualInput').value = cleanVal; searchRecord(cleanVal); stopScanner(); }, (err) => {} ); isScanning = true; document.getElementById('scanBtn').innerText = "Stop Scanner"; } } catch (err) { console.error("Camera launch failed:", err); alert("Camera shuru nahi ho saka: " + (err.message || err)); readerDiv.classList.add('hidden'); readerDiv.style.display = 'none'; isScanning = false; } } else { stopScanner(); } }); function stopScanner() { if (html5QrCode && isScanning) { html5QrCode.stop().then(() => { const readerDiv = document.getElementById('reader'); readerDiv.classList.add('hidden'); readerDiv.style.display = 'none'; isScanning = false; document.getElementById('scanBtn').innerHTML = `Open High-Speed Scanner`; }).catch(() => { const readerDiv = document.getElementById('reader'); readerDiv.classList.add('hidden'); readerDiv.style.display = 'none'; isScanning = false; document.getElementById('scanBtn').innerHTML = `Open High-Speed Scanner`; }); } }
